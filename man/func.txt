{yellow}NAME{reset}
    {cyan}func{reset} - Package manager for cSH functions.

{yellow}SYNOPSIS{reset}
    {cyan}func{reset} [{cyan}command{reset}] [{green}target{reset}] [{magenta}options{reset}]

    Commands: {cyan}install{reset} | {cyan}uninstall{reset} | {cyan}list{reset} | {cyan}update{reset} | {cyan}full-upgrade{reset} | 
              {cyan}search{reset} | {cyan}add-repo{reset} | {cyan}remove-repo{reset} | {cyan}remake-config{reset} | {cyan}help{reset}

{yellow}COMMANDS{reset}
    {cyan}install{reset} {green}<pkg>{reset}       Resolves dependencies and installs the specified package.
    {cyan}uninstall{reset} {green}<pkg>{reset}     Removes an installed package from your local system.
                        (Alias: {cyan}remove{reset})
    {cyan}list{reset}                Lists all available packages in the cached repositories.
    {cyan}list{reset} {magenta}--installed{reset}    Lists only the packages currently installed on your system.
    {cyan}update{reset}              Refreshes the local repository indexes by fetching the latest data.
                        (Alias: {cyan}update-source{reset})
    {cyan}full-upgrade{reset}        Checks all installed packages and re-downloads them to ensure 
                        they are on the latest cached versions. (Alias: {cyan}upgrade{reset})
    {cyan}search{reset} {green}<keyword>{reset}    Searches cached repositories for a specific term.
    {cyan}add-repo{reset} {green}<url>{reset}      Adds a new repository URL to your configuration file.
    {cyan}remove-repo{reset} {green}<url>{reset}   Removes a specified repository URL from your configuration file.
    {cyan}remake-config{reset}       Resets the configuration file to the default repository list.
    {cyan}help{reset} [{green}command{reset}]      Displays the general help menu or detailed usage.

{yellow}OPTIONS{reset}
    {magenta}-h{reset}, {magenta}--help{reset}          Can be appended to any command to display specific usage instructions.
