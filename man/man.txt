{yellow}NAME{reset}
    {cyan}man{reset} - Manual page manager and viewer.

{yellow}SYNOPSIS{reset}
    {cyan}man{reset} [{cyan}install{reset} | {cyan}update{reset} | {cyan}list{reset} | {cyan}remove{reset} | {cyan}purge{reset} | {green}<command>{reset}]

{yellow}COMMANDS{reset}
    {green}<command>{reset}        Views the manual page for the specified command.
    {cyan}install{reset} {green}<name>{reset}   Downloads a manual page from the repository.
    {cyan}update{reset}           Refreshes the manual repository index.
    {cyan}list{reset}             Lists all locally installed manual pages.
    {cyan}remove{reset} {green}<name>{reset}    Deletes a specific manual page.
    {cyan}purge{reset}            Deletes all manual pages and configs.
